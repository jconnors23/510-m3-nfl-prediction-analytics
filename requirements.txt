# Every package is pinned to an exact version (==) so any machine installs the same thing.
# Install with: pip install -r requirements.txt
# Python 3.13 is needed. The Lambda function in Phase 4 and the Docker image in Phase 6 must use the same minor version.

# Data tables and math
# pandas holds the game tables and does the filtering and grouping. numpy does the number
# work underneath it. PyYAML reads config/config.yaml.
pandas==3.1.1
numpy==2.5.4
PyYAML==6.0.3

# Model
# scikit-learn supplies the standardizing step and the logistic regression, and the scores
# (accuracy and log loss). The API in Phase 3 must use this same version, because a saved
# model may fail to load, or answer differently, under another one.
scikit-learn==1.9.1
# scikit-learn's solver uses scipy while fitting, so an unpinned scipy could fit slightly different weights on another machine.
scipy==1.18.1

# Experiment tracking
# mlflow records the settings, scores, and files of each training run (Phase 2).
mlflow==3.18.0

# Training outputs
# matplotlib draws the confusion-matrix image. joblib writes and reads the saved model file.
matplotlib==3.11.2
joblib==1.6.0

# Data download
# requests downloads the games file in the fetch step (Phase 1).
requests==2.34.2

# Cloud storage access (Amazon S3)
# boto3 is the Python library for talking to AWS from code. It uploads the cleaned table and
# downloads it back (Phase 1), and the feedback handler uses it to write to DynamoDB.
boto3==1.44.5

# Model serving API (Phases 3 and 4)
# fastapi builds the /predict route, pydantic checks that each request has the right fields,
# uvicorn runs the API on a local machine, and mangum lets AWS Lambda call the same app.
fastapi==0.162.2
uvicorn[standard]==0.54.0
pydantic==2.14.3
mangum==0.22.0

# Tests
# pytest runs everything in tests/. The tests use made-up data and a stand-in S3 client, so
# they need no internet and no AWS account.
pytest==9.2.0
# httpx2 is used only by the API tests (Starlette's test client asks for it), and it is not needed in the Lambda zip.
httpx2==2.16.2
